function currentList(){ var f = $("filter").value; return f ? data.filter(function(i){ return i.deviceId === f; }) : data; } function renderFilter(){ var sel = $("filter"), keep = sel.value; var ids = Array.from(new Set(data.map(function(i){ return i.deviceId; }))).sort(); sel.textContent = ""; sel.appendChild(el("option", {value:""}, "All devices")); ids.forEach(function(id){ sel.appendChild(el("option", {value:id}, id)); }); sel.value = ids.indexOf(keep) >= 0 ? keep : ""; } function renderSummary(){ var list = currentList(); $("s-total").textContent = list.length; $("s-devices").textContent = new Set(list.map(function(i){ return i.deviceId; })).size; $("s-crit").textContent = list.filter(function(i){ return i.status === "CRITICAL"; }).length; $("s-latest").textContent = list.length ? list[0].metricValue : "–"; } function renderTable(){ var body = $("rows"), list = currentList(); body.textContent = ""; if (!list.length){ var tr = el("tr"), td = el("td", {colspan:"5", "class":"empty"}, "No readings yet. Send a test reading to see it here."); tr.appendChild(td); body.appendChild(tr); return; } list.slice(0, 50).forEach(function(i){ var tr = el("tr"); tr.appendChild(el("td", null, fmtTime(i.timestamp))); tr.appendChild(el("td", null, i.deviceId)); tr.appendChild(el("td", null, i.metricType)); tr.appendChild(el("td", {"class":"num"}, String(i.metricValue))); var s = el("td"); s.appendChild(el("span", {"class":"badge " + (i.status === "CRITICAL" ? "CRITICAL" : "")}, i.status === "CRITICAL" ? "Critical" : "Healthy")); tr.appendChild(s); body.appendChild(tr); }); } function renderChart(){ var svg = $("chart"); svg.textContent = ""; var pts = currentList().slice(0, 40).reverse(); var W = 640, H = 260, L = 40, R = 14, T = 14, B = 28; if (!pts.length){ svg.appendChild(svgEl("text", {x:W/2, y:H/2, "text-anchor":"middle"}, "No data yet")); return; } var vals = pts.map(function(p){ return p.metricValue; }); var max = Math.max(THRESHOLD + 5, Math.max.apply(null, vals) * 1.05); var min = Math.min(0, Math.min.apply(null, vals)); var x = function(i){ return pts.length === 1 ? (L + (W-L-R)/2) : L + i * (W-L-R) / (pts.length - 1); }; var y = function(v){ return T + (1 - (v - min) / (max - min)) * (H-T-B); }; [0, .25, .5, .75, 1].forEach(function(f){ var v = min + f * (max - min), yy = y(v); svg.appendChild(svgEl("line", {x1:L, x2:W-R, y1:yy, y2:yy, stroke:"var(--line)", "stroke-width":1})); svg.appendChild(svgEl("text", {x:L-6, y:yy+4, "text-anchor":"end"}, String(Math.round(v)))); }); svg.appendChild(svgEl("line", {x1:L, x2:W-R, y1:y(THRESHOLD), y2:y(THRESHOLD), stroke:"var(--crit)", "stroke-width":1.5, "stroke-dasharray":"6 4"})); svg.appendChild(svgEl("text", {x:W-R, y:y(THRESHOLD)-6, "text-anchor":"end", style:"fill:var(--crit)"}, "Critical above " + THRESHOLD)); svg.appendChild(svgEl("polyline", { points: pts.map(function(p, i){ return x(i) + "," + y(p.metricValue); }).join(" "), fill:"none", stroke:"var(--teal)", "stroke-width":2, "stroke-linejoin":"round" })); pts.forEach(function(p, i){ var c = svgEl("circle", {cx:x(i), cy:y(p.metricValue), r: p.status === "CRITICAL" ? 5 : 3.5, fill: p.status === "CRITICAL" ? "var(--crit)" : "var(--teal)"}); c.appendChild(svgEl("title", {}, p.deviceId + ": " + p.metricValue + " at " + fmtTime(p.timestamp))); svg.appendChild(c); }); svg.appendChild(svgEl("text", {x:L, y:H-8}, fmtTime(pts[0].timestamp))); svg.appendChild(svgEl("text", {x:W-R, y:H-8, "text-anchor":"end"}, fmtTime(pts[pts.length-1].timestamp))); } function renderAll(){ renderFilter(); renderSummary(); renderTable(); renderChart(); } function load(){ return fetch(API, {headers:{"Accept":"application/json"}}) .then(function(r){ if (!r.ok) throw new Error("HTTP " + r.status); return r.json(); }) .then(function(items){ data = items; renderAll(); setLive(true, "Updated " + new Date().toLocaleTimeString()); }) .catch(function(e){ setLive(false, "Can't reach the API (" + e.message + ")"); }); } $("filter").addEventListener("change", function(){ renderSummary(); renderTable(); renderChart(); }); $("send").addEventListener("click", function(){ var btn = $("send"), msg = $("send-msg"); msg.className = "msg"; msg.textContent = ""; var body = { deviceId: $("f-device").value.trim(), metricType: $("f-type").value, metricValue: parseFloat($("f-value").value) }; if (!body.deviceId){ msg.className = "msg err"; msg.textContent = "Enter a device ID."; return; } if (isNaN(body.metricValue)){ msg.className = "msg err"; msg.textContent = "Enter a numeric value."; return; } btn.disabled = true; fetch(API, {method:"POST", headers:{"Content-Type":"application/json"}, body:JSON.stringify(body)}) .then(function(r){ return r.json().then(function(j){ return {ok:r.ok, j:j}; }); }) .then(function(res){ if (!res.ok) throw new Error(res.j.error || "Request failed"); msg.textContent = res.j.item.status === "CRITICAL" ? (res.j.alertSent ? "Sent. Critical reading, alert dispatched." : "Sent. Critical reading, but the alert failed.") : "Sent. Reading is healthy."; return load(); }) .catch(function(e){ msg.className = "msg err"; msg.textContent = e.message; }) .then(function(){ btn.disabled = false; }); }); load(); setInterval(load, REFRESH_MS); })();
